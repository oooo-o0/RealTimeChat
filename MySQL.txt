-- データベースの作成
CREATE DATABASE IF NOT EXISTS chat_system_db;

-- 使用するデータベースの指定
USE chat_system_db;

-- user テーブルの作成
CREATE TABLE `user` (
    `id` INT NOT NULL AUTO_INCREMENT,        -- ユーザーID (AUTO_INCREMENT)
    `name` VARCHAR(255) NOT NULL,            -- ユーザー名
    `email` VARCHAR(255) NOT NULL,           -- メールアドレス
    `profile` VARCHAR(255),                  -- プロフィール情報
    `password` VARCHAR(255) NOT NULL,        -- パスワード（ハッシュ化されている）
    PRIMARY KEY (`id`),                      -- 主キー設定
    UNIQUE KEY `email` (`email`)             -- メールアドレスは一意
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- chat テーブルの作成
CREATE TABLE `chat` (
    `id` INT NOT NULL AUTO_INCREMENT,        -- チャットID (AUTO_INCREMENT)
    `chat_name` VARCHAR(255) DEFAULT NULL,   -- チャット名 (NULLを許容)
    `chat_image` VARCHAR(255) DEFAULT NULL,  -- チャットアイコンURL (NULLを許容)
    `is_group` BOOLEAN NOT NULL,             -- グループチャットかどうか
    `created_by` INT NOT NULL,               -- 作成者ID (外部キー)
    PRIMARY KEY (`id`),
    FOREIGN KEY (`created_by`) REFERENCES `user`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- message テーブルの作成
CREATE TABLE `message` (
    `id` INT NOT NULL AUTO_INCREMENT,        -- メッセージID (AUTO_INCREMENT)
    `content` TEXT NOT NULL,                 -- メッセージ内容
    `timestamp` DATETIME NOT NULL,           -- 送信日時
    `chat_id` INT NOT NULL,                  -- チャットID (外部キー)
    `user_id` INT NOT NULL,                  -- 送信者ID (外部キー)
    PRIMARY KEY (`id`),
    FOREIGN KEY (`chat_id`) REFERENCES `chat`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`user_id`) REFERENCES `user`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Chat_Admins テーブルの作成
CREATE TABLE `chat_admins` (
    `chat_id` INT NOT NULL,                   -- チャットID (外部キー)
    `user_id` INT NOT NULL,                   -- ユーザーID (外部キー)
    PRIMARY KEY (`chat_id`, `user_id`),
    FOREIGN KEY (`chat_id`) REFERENCES `chat`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`user_id`) REFERENCES `user`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Chat_Users テーブルの作成
CREATE TABLE `chat_users` (
    `chat_id` INT NOT NULL,                   -- チャットID (外部キー)
    `user_id` INT NOT NULL,                   -- ユーザーID (外部キー)
    PRIMARY KEY (`chat_id`, `user_id`),
    FOREIGN KEY (`chat_id`) REFERENCES `chat`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`user_id`) REFERENCES `user`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
-- user_seq テーブルの作成
CREATE TABLE `user_seq` (
    `next_val` INT NOT NULL AUTO_INCREMENT,
    PRIMARY KEY (`next_val`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- chat_seq テーブルの作成
CREATE TABLE `chat_seq` (
    `next_val` INT NOT NULL AUTO_INCREMENT,
    PRIMARY KEY (`next_val`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- message_seq テーブルの作成
CREATE TABLE `message_seq` (
    `next_val` INT NOT NULL AUTO_INCREMENT,
    PRIMARY KEY (`next_val`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
